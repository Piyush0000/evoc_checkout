Quick Setup

To use on your website, add the SDK script and a container element: 

<script src="evoc-checkout-sdk.js"></script>

<div id="evoc-checkout"></div>

<script>
  EVOC_CHECKOUT.init({
    storeId: 'your_store_id',
    apiBaseUrl: 'https://your-api.com/api/v1',
    amount: 4999,
    onSuccess: (order) => {
      console.log('Order placed!', order);
    }
  });
</script>
